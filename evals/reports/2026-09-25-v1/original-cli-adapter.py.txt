"""Direct Claude Code CLI comparison; uses the frozen suite's grader."""
import argparse
from concurrent.futures import ThreadPoolExecutor, as_completed
import importlib.util
import json
import os
from pathlib import Path
import random
import signal
import subprocess
import time

parser=argparse.ArgumentParser()
parser.add_argument('--suite',type=Path,required=True)
parser.add_argument('--output',type=Path,required=True)
parser.add_argument('--cases')
parser.add_argument('--repetitions',type=int,default=3)
parser.add_argument('--jobs',type=int,default=3)
parser.add_argument('--model',default='claude-sonnet-5')
parser.add_argument('--seconds',type=int,default=300)
parser.add_argument('--max-turns',type=int,default=40)
parser.add_argument('--attempts',type=int,default=1)
parser.add_argument('--seed',type=int,default=42)
args=parser.parse_args()
suite=args.suite.resolve()
spec=importlib.util.spec_from_file_location('evals_runner',suite/'run.py')
bench=importlib.util.module_from_spec(spec); spec.loader.exec_module(bench)
output=args.output.resolve(); output.mkdir(parents=True,exist_ok=False)
cases=sorted(p for p in (suite/'cases').iterdir() if p.is_dir())
if args.cases: cases=[p for p in cases if p.name in args.cases.split(',')]
bench.validate_cases(cases,output/'validation')
paths=[p for p in suite.rglob('*') if p.is_file() and 'results' not in p.parts and '__pycache__' not in p.parts]
hashes={str(p.relative_to(suite)):bench.digest(p) for p in paths}
bench.write_json(output/'manifest.json',{'suite_sha256':hashes,
    'adapter_sha256':bench.digest(Path(__file__)),
    'arguments':{k:str(v) if isinstance(v,Path) else v for k,v in vars(args).items()},
    'claude_version':subprocess.check_output(['claude','--version'],text=True).strip(),
    'system_prompt':'Claude Code default; no --system-prompt override'})

def trial(job):
    case,rep=job
    directory=output/f'{case.name}--claude_cli--{rep}'
    spec=bench.prepare_trial(case,'raw',rep,args,directory)
    spec['mode']='claude_cli'; bench.write_json(directory/'spec.json',spec)
    workspace=directory/'workspace'
    cmd=['claude','-p','--output-format','stream-json','--verbose',
         '--model',args.model,'--max-turns',str(args.max_turns),
         '--tools',','.join(bench.TOOLS),'--allowedTools',','.join(bench.TOOLS),
         '--permission-mode','dontAsk','--setting-sources','','--strict-mcp-config']
    prompt=spec['task']+'\n\n'+bench.CONSTRAINTS
    started=time.monotonic(); error=None
    with (directory/'transcript.jsonl').open('w') as stdout, (directory/'stderr.log').open('w') as stderr:
        process=subprocess.Popen(cmd,cwd=workspace,stdin=subprocess.PIPE,stdout=stdout,stderr=stderr,text=True,
                    start_new_session=True,env={**os.environ,'CLAUDE_CODE_DISABLE_BACKGROUND_TASKS':'1'})
        try: process.communicate(prompt,timeout=args.seconds)
        except subprocess.TimeoutExpired:
            os.killpg(process.pid,signal.SIGKILL); process.communicate(); error='Whole-trial timeout'
    elapsed=time.monotonic()-started
    messages=[]
    for line in (directory/'transcript.jsonl').read_text().splitlines():
        try: message=json.loads(line)
        except ValueError: continue
        if message.get('type')=='result': messages.append(message)
    final=messages[-1] if messages else {}
    completed=bool(process.returncode==0 and final.get('subtype')=='success' and not final.get('is_error')
                   and not final.get('permission_denials') and final.get('result'))
    if not completed and not error: error=f'CLI failed: exit={process.returncode}, subtype={final.get("subtype")}'
    score=bench.grade(case,workspace,directory/'acceptance')
    subprocess.run(['git','add','-N','.'],cwd=workspace,check=True)
    diff=subprocess.check_output(['git','diff','HEAD','--','.'],cwd=workspace,text=True)
    (directory/'changes.diff').write_text(diff)
    usages=[m.get('usage') or {} for m in messages]
    result={**spec,'completed':completed,'error':error,'elapsed_seconds':elapsed,'wall_seconds':elapsed,
            'acceptance':score,'accepted':bool(completed and score['passed']),
            'false_completion':bool(completed and not score['passed']),
            'usage':{k:sum(u.get(k,0) for u in usages) for k in
                     ['input_tokens','output_tokens','cache_read_input_tokens','cache_creation_input_tokens']},
            'usage_complete':bool(messages) and completed,
            'list_price_usd':sum(m.get('total_cost_usd') or 0 for m in messages),
            'resolved_models':sorted({k for m in messages for k in m.get('modelUsage',{})}),
            'agent_calls':len(messages),'worker_attempts':1,'human_interventions':0,'human_review':None}
    bench.write_json(directory/'result.json',result)
    print(f'{directory.name}: accepted={result["accepted"]} time={elapsed:.1f}s',flush=True)

jobs=[(c,r) for c in cases for r in range(1,args.repetitions+1)]
random.Random(args.seed).shuffle(jobs)
bench.write_json(output/'schedule.json',[(c.name,'claude_cli',r) for c,r in jobs])
with ThreadPoolExecutor(max_workers=args.jobs) as pool:
    for future in as_completed([pool.submit(trial,j) for j in jobs]): future.result()
changed=[str(p.relative_to(suite)) for p in paths if not p.exists() or bench.digest(p)!=hashes[str(p.relative_to(suite))]]
bench.write_json(output/'integrity.json',{'unchanged':not changed,'changed':changed})
if changed: raise SystemExit('Suite changed during execution')
